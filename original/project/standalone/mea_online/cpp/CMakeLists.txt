cmake_minimum_required(VERSION 3.16)
project(mea_online_collector LANGUAGES CXX)
option(MEA_SIMULATION "Build explicit test-only fake SDK producer" OFF)
set(MAXLAB_ROOT "" CACHE PATH "MaxLab root containing include/maxlab and lib/libmaxlab.a")
set(MAXLAB_CXX_RUNTIME "" CACHE FILEPATH "Optional newer libstdc++.so.6 compatible with target glibc")
add_executable(mea_window_capture mea_window_capture.cpp)
target_compile_features(mea_window_capture PRIVATE cxx_std_17)
target_compile_options(mea_window_capture PRIVATE -Wall -Wextra -Werror)
if(MEA_SIMULATION)
  target_include_directories(mea_window_capture PRIVATE stub)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/stub/maxlab/maxlab.h" header_sha)
  set(library_sha "${header_sha}")
  target_compile_definitions(mea_window_capture PRIVATE MEA_SIMULATION)
else()
  if(NOT EXISTS "${MAXLAB_ROOT}/include/maxlab/data_streamer.h" OR NOT EXISTS "${MAXLAB_ROOT}/lib/libmaxlab.a")
    message(FATAL_ERROR "Supply actual MaxLab >=1.1 filtered-frameInfo SDK via MAXLAB_ROOT")
  endif()
  file(SHA256 "${MAXLAB_ROOT}/include/maxlab/data_streamer.h" header_sha)
  file(SHA256 "${MAXLAB_ROOT}/lib/libmaxlab.a" library_sha)
  target_include_directories(mea_window_capture PRIVATE "${MAXLAB_ROOT}/include")
  target_link_libraries(mea_window_capture PRIVATE "${MAXLAB_ROOT}/lib/libmaxlab.a")
  if(MAXLAB_CXX_RUNTIME)
    target_link_libraries(mea_window_capture PRIVATE "${MAXLAB_CXX_RUNTIME}")
  endif()
endif()
target_compile_definitions(mea_window_capture PRIVATE MEA_HEADER_SHA256="${header_sha}" MEA_LIBRARY_SHA256="${library_sha}")
target_link_libraries(mea_window_capture PRIVATE pthread)
